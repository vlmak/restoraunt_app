--автоматическая отмена брони если гости не пришли
CREATE OR REPLACE FUNCTION public.auto_cancel_expired_reservations()
 RETURNS void
 LANGUAGE plpgsql
AS $function$
DECLARE
    rec RECORD;
    v_other_active INT;
BEGIN
    -- Находим все активные брони, где время брони прошло более 30 минут назад
    FOR rec IN (
        SELECT reservation_id, table_id
        FROM reservations
        WHERE status = 'active'
          AND reservation_time < NOW() - INTERVAL '30 minutes'
    ) LOOP
        -- Обновляем статус брони на 'no_show'
        UPDATE reservations
        SET status = 'no_show'
        WHERE reservation_id = rec.reservation_id;
        
        -- Проверяем, есть ли другие активные брони на этот столик
        SELECT COUNT(*) INTO v_other_active
        FROM reservations
        WHERE table_id = rec.table_id
          AND status IN ('active', 'arrived')
          AND reservation_id != rec.reservation_id;
        
        -- Если нет других активных броней, освобождаем столик
        IF v_other_active = 0 THEN
            UPDATE tables
            SET status = 'free'
            WHERE table_id = rec.table_id
              AND status = 'reserved';  -- Освобождаем только если столик всё ещё зарезервирован
        END IF;
        
        RAISE NOTICE 'Бронь % автоматически отменена (no_show)', rec.reservation_id;
    END LOOP;
END;
$function$
;



--перевод столика в статус "свободен" после уборки
CREATE OR REPLACE FUNCTION public.auto_clean_tables()
 RETURNS void
 LANGUAGE plpgsql
AS $function$
BEGIN
    UPDATE tables
    SET 
        status = 'free',
        cleaning_started_at = NULL
    WHERE status = 'needs_cleaning'
      AND cleaning_started_at < NOW() - INTERVAL '5 minutes';
END;
$function$
;

--проверка столика перед бронью
CREATE OR REPLACE FUNCTION public.check_table_availability_before_reservation()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_capacity INT;
    v_status VARCHAR(20);
    v_conflict_count INT;
BEGIN
    -- 1. Проверяем, существует ли столик с указанным ID
    SELECT capacity, status INTO v_capacity, v_status
    FROM tables
    WHERE table_id = NEW.table_id;
    
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Столик с ID % не существует', NEW.table_id;
    END IF;
    
    -- 2. Проверяем, достаточно ли вместимости
    IF v_capacity < NEW.persons THEN
        RAISE EXCEPTION 'Столик % вмещает только % человек, а вы указали %', 
            NEW.table_id, v_capacity, NEW.persons;
    END IF;
    
    -- 3. Проверяем, не занят ли столик
    IF v_status = 'occupied' THEN
        RAISE EXCEPTION 'Столик % уже занят гостями', NEW.table_id;
    END IF;
    
    -- 4. Проверяем, не требует ли столик уборки
    IF v_status = 'needs_cleaning' THEN
        RAISE EXCEPTION 'Столик % требует уборки. Попробуйте позже.', NEW.table_id;
    END IF;
    
    -- 5. Проверяем, нет ли других активных броней на это время
    --    (считаем, что бронь может пересекаться, если резервируется на то же время)
    SELECT COUNT(*) INTO v_conflict_count
    FROM reservations
    WHERE table_id = NEW.table_id
      AND status IN ('active', 'arrived')
      AND reservation_time = NEW.reservation_time;
    
    IF v_conflict_count > 0 THEN
        RAISE EXCEPTION 'Столик % уже забронирован на это время', NEW.table_id;
    END IF;
    
    RETURN NEW;
END;
$function$
;


--установка цены ингредиента для заказа
CREATE OR REPLACE FUNCTION public.set_supplier_order_item_price()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    -- Если цена не указана явно — подставляем из ingredients
    IF NEW.unit_price IS NULL OR NEW.unit_price = 0 THEN
        SELECT price INTO NEW.unit_price
        FROM ingredients
        WHERE ingredient_id = NEW.ingredient_id;
    END IF;
    
    RETURN NEW;
END;
$function$
;


--подсчет цены чека
CREATE OR REPLACE FUNCTION public.update_check_total()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    -- Обновляем сумму в таблице checks
    UPDATE checks
    SET total_amount = (
        SELECT COALESCE(SUM(price_at_moment), 0)
        FROM check_items
        WHERE check_id = NEW.check_id
          AND status = 'ready'
    )
    WHERE check_id = NEW.check_id;
    
    RETURN NEW;
END;
$function$
;

--подсчет цены заказанных ингредиентов
CREATE OR REPLACE FUNCTION public.update_supplier_order_total()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_order_id INT;
BEGIN
    -- Определяем ID заказа
    IF TG_OP = 'DELETE' THEN
        v_order_id = OLD.order_id;
    ELSE
        v_order_id = NEW.order_id;
    END IF;

    -- Обновляем сумму заказа
    UPDATE supplier_orders
    SET total_amount = (
        SELECT COALESCE(SUM(quantity * unit_price), 0)
        FROM supplier_order_items
        WHERE order_id = v_order_id
    )
    WHERE order_id = v_order_id;

    RETURN NULL;
END;
$function$
;

--перевод столика в статус "требует уборки"
CREATE OR REPLACE FUNCTION public.update_table_status_on_check_close()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    IF OLD.status IS DISTINCT FROM NEW.status AND NEW.status = 'closed' THEN
        UPDATE tables 
        SET 
            status = 'needs_cleaning',
            cleaning_started_at = NOW()
        WHERE table_id = NEW.table_id;
    END IF;
    
    RETURN NEW;
END;
$function$
;


--бронь столика
CREATE OR REPLACE FUNCTION public.update_table_status_on_reservation()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_other_active INT;
BEGIN
    -- Если статус брони изменился или новая запись
    IF TG_OP = 'INSERT' OR OLD.status IS DISTINCT FROM NEW.status THEN
        
        -- 1. Если бронь активна столик резервируется
        IF NEW.status = 'active' THEN
            UPDATE tables 
            SET status = 'reserved' 
            WHERE table_id = NEW.table_id;
            
        -- 2. Если гости пришли столик занимается
        ELSIF NEW.status = 'arrived' THEN
            UPDATE tables 
            SET status = 'occupied' 
            WHERE table_id = NEW.table_id;
            
        -- 3. Если бронь отменена или гости не пришли
        ELSIF NEW.status IN ('cancelled', 'no_show') THEN
            -- Проверяем, есть ли другие активные брони на этот столик
            SELECT COUNT(*) INTO v_other_active
            FROM reservations
            WHERE table_id = NEW.table_id
              AND status IN ('active', 'arrived')
              AND reservation_id != NEW.reservation_id;
            
            -- Если нет других активных броней, освобождаем столик
            IF v_other_active = 0 THEN
                UPDATE tables 
                SET status = 'free' 
                WHERE table_id = NEW.table_id;
            END IF;
        END IF;
    END IF;
    
    RETURN NEW;
END;
$function$
;

--автоматическое открытие чека после прихода гостей
CREATE OR REPLACE FUNCTION public.update_table_status_on_check_open()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_table_status VARCHAR(20);
BEGIN
    -- Проверяем текущий статус столика
    SELECT status INTO v_table_status
    FROM tables
    WHERE table_id = NEW.table_id;
    
    -- Нельзя открыть чек, если столик требует уборки
    IF v_table_status = 'needs_cleaning' THEN
        RAISE EXCEPTION 'Столик % требует уборки. Невозможно открыть чек.', NEW.table_id;
    END IF;
    
    -- Нельзя открыть чек, если столик уже занят
    IF v_table_status = 'occupied' THEN
        RAISE EXCEPTION 'Столик % уже занят', NEW.table_id;
    END IF;
    
    -- Обновляем статус столика на occupied
    UPDATE tables 
    SET status = 'occupied' 
    WHERE table_id = NEW.table_id;
    
    -- Если у столика есть активная бронь, обновляем её статус на 'arrived'
    UPDATE reservations 
    SET status = 'arrived' 
    WHERE table_id = NEW.table_id
      AND status = 'active'
      AND reservation_time <= NOW();  -- Бронь считается актуальной, если время уже наступило
    
    RETURN NEW;
END;
$function$
;


--ТРИГГЕРЫ
create trigger trg_update_check_total after
insert
    or
delete
    or
update
    of status on
    public.check_items for each row execute function update_check_total();

create trigger trg_update_table_status_on_check_open after
insert
    on
    public.checks for each row execute function update_table_status_on_check_open();

create trigger trg_update_table_status_on_check_close after
update
    of status on
    public.checks for each row
    when (((new.status)::text = 'closed'::text)) execute function update_table_status_on_check_close();

create trigger trg_check_reservation_availability before
insert
    on
    public.reservations for each row execute function check_table_availability_before_reservation();

create trigger trg_update_table_status_on_reservation after
insert
    or
update
    of status on
    public.reservations for each row execute function update_table_status_on_reservation();

create trigger trg_update_supplier_order_total after
insert
    or
delete
    or
update
    on
    public.supplier_order_items for each row execute function update_supplier_order_total();

create trigger trg_set_supplier_order_item_price before
insert
    on
    public.supplier_order_items for each row execute function set_supplier_order_item_price();
